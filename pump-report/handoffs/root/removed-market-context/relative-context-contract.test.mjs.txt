import test from 'node:test';import assert from 'node:assert/strict';
import{qualifyRelativeContext,RELATIVE_CONTEXT_SCHEMA,RELATIVE_CONTEXT_UNIT,RELATIVE_PRICE_BASIS}from'../../contracts/relative-context.mjs';
import{CANDLE_CCXT_TIME_BASIS}from'../../contracts/index.mjs';
const T=Date.parse('2026-10-25T12:00:00Z');
const instrument=base=>({instrument_id:'BINGX:USDT_M:ARTIFICIAL:'+base,asset_id:'ARTIFICIAL:'+base,native_symbol:base+'-USDT',base,venue:'BINGX',market_type:'PERPETUAL',quote:'USDT',settle:'USDT',classification_status:'VERIFIED_CRYPTO',identity_evidence_ref:'ARTIFICIAL_ONLY'});
const candidate=instrument('人工TEST');
const leg=(role,identity)=>({role,instrument:identity,start_ts:T-300000,end_ts:T,start_price:100,end_price:110,return_pct:100*(110/100-1),fact_id:'ARTIFICIAL_CLOSED:'+role,price_basis:RELATIVE_PRICE_BASIS,time_basis:CANDLE_CCXT_TIME_BASIS,interval_ms:60000,complete_closed_window:true});
const fixture=()=>({fact_id:'ARTIFICIAL_CONTEXT',instrument_id:candidate.instrument_id,kind:'market_context',value:{schema_version:RELATIVE_CONTEXT_SCHEMA,returns:[leg('CANDIDATE',candidate),leg('BTC',instrument('BTC')),leg('ETH',instrument('ETH'))]},status:'USABLE',origin:'OBSERVED',venue:'BINGX',unit:RELATIVE_CONTEXT_UNIT,event_ts:T,received_ts:T+1000,expires_ts:T+90000});
test('qualified equal-window native returns preserve explicit Unicode candidate and benchmark identities',()=>{const r=qualifyRelativeContext(fixture(),{instrument:candidate,as_of_ts:T+50000});assert.equal(r.valid,true);assert.equal(r.coverage,'BTC_AND_ETH');assert.equal(r.independent_venue_count,1);const f=fixture();f.value.returns.pop();assert.equal(qualifyRelativeContext(f,{instrument:candidate,as_of_ts:T+50000}).coverage,'ONE_BENCHMARK');});
test('wrong benchmark, mismatched clocks or calculation and future receipt cannot create relative strength',()=>{for(const patch of [f=>{f.value.returns[1].instrument=instrument('ETH');},f=>{f.value.returns[1].start_ts-=60000;},f=>{f.value.returns[1].return_pct=42;},f=>{f.value.returns[1].complete_closed_window=false;},f=>{f.received_ts=T+60000;},f=>{f.value.returns[1].instrument.classification_status='UNKNOWN';},f=>{delete f.value.schema_version;},f=>{f.expires_ts=T+90001;}]){const f=fixture();patch(f);assert.equal(qualifyRelativeContext(f,{instrument:candidate,as_of_ts:T+50000}).valid,false);}});
