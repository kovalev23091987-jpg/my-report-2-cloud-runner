  need(APPROVED_HTX_SCOPE.approved_assets===105&&APPROVED_HTX_SCOPE.approved_contracts===122&&APPROVED_HTX_ANALYSIS_CONTRACTS.length===105&&APPROVED_HTX_NATIVE_CONTRACTS.length===122,'INDEPENDENT_APPROVED_ROSTER_REQUIRED');
  const rows=APPROVED_HTX_ANALYSIS_CONTRACTS;
  need(new Set(rows.map(x=>x.contract_code)).size===105&&rows.every(x=>x.contract_code===x.asset_symbol+'-USDT'),'INDEPENDENT_EXACT_ANALYSIS_KEYS_REQUIRED');
  return{schema:'INDEPENDENT_ANOMALY_CAPTURE_ROSTER_V1',generation:APPROVED_HTX_SCOPE.generation,manifest_sha256:APPROVED_HTX_SCOPE.manifest_sha256,roster_sha256:digest({scope:APPROVED_HTX_SCOPE,analysis:rows,native:APPROVED_HTX_NATIVE_CONTRACTS}),analysis_assets:105,native_contracts:122,unsupported_other_native_contracts:17,structural_membership_only:true,current_useful_coverage:null};
}

      pending=Promise.resolve().then(()=>{need(Date.now()<deadline_ts,'INDEPENDENT_IO_DEADLINE_BEFORE_ACTUAL_FETCH');calls++;return fetch_json(url);}).then(x=>{value=x;return x;},e=>{failed=e;throw e;});return pending;
