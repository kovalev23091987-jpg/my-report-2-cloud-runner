// Same runner filesystem seam; private local continuity is NOT native ACK.
const RUNNER_ANOMALY_PENDING_JOURNALS=new WeakSet();
async function preflightRunnerAnomalyPendingJournal({file,expected_sha256,generation,source_head,cloud_run_id}) {
  const need=(v,m)=>{if(!v)throw Error(m);};
  need(typeof generation==='string'&&generation.length>0&&generation.length<=160&&/^[a-f0-9]{40}$/.test(source_head||'')&&/^[1-9]\d{0,15}$/.test(cloud_run_id||''),'ANOMALY_JOURNAL_CURRENT_EXECUTION_IDENTITY_REQUIRED');
  if(expected_sha256==='INITIALIZE_EMPTY'){
    for(const name of [file,file+'.next',file+'.step-001'])try{await fs.lstat(name);throw Error('ANOMALY_JOURNAL_TRUSTED_INIT_TARGET_NOT_EMPTY');}catch(e){if(e.code!=='ENOENT')throw e;}
    return{status:'EXPLICIT_EMPTY_TARGET_CHECKED_NOT_CREATED',SQL:0,native_admission:false};
  }
  need(typeof expected_sha256==='string'&&/^[a-f0-9]{64}$/.test(expected_sha256),'ANOMALY_JOURNAL_INDEPENDENT_EXPECTED_SHA_REQUIRED');
  const st=await fs.lstat(file);need(st.isFile()&&!st.isSymbolicLink()&&(st.mode&0o077)===0&&st.size<=16*1048576+65536&&await fs.realpath(file)===file,'ANOMALY_JOURNAL_REAL_FILE_REQUIRED');
  const text=await fs.readFile(file,'utf8');need(Buffer.byteLength(text,'utf8')===st.size&&crypto.createHash('sha256').update(text).digest('hex')===expected_sha256,'ANOMALY_JOURNAL_EXPECTED_SHA_CONFLICT');
  // Identity is read only AFTER independent full-byte expectation matches.
  const identity=JSON.parse(text).identity;
  need(identity?.generation===generation&&identity.source_head===source_head&&identity.cloud_run_id===cloud_run_id,'ANOMALY_JOURNAL_CURRENT_EXECUTION_IDENTITY_CONFLICT');
  const journal=await openRunnerAnomalyPendingJournal({file,identity,expected_sha256});
  need(!journal.blocked(),'ANOMALY_PENDING_RESTART_STOPS_BEFORE_NEW_PARENT_SQL');
  return{status:'CLOSED_PRIOR_JOURNAL_PRECHECKED',checkpoint:journal.checkpoint(),SQL:0,native_admission:false};
}
async function openRunnerAnomalyPendingJournal({file,identity,expected_sha256,initialize_empty=false}) {
  const need=(v,m)=>{if(!v)throw Error(m);},hex=x=>typeof x==='string'&&/^[a-f0-9]{64}$/.test(x);
  const canon=canonicalAnomalyJSON,hash=s=>crypto.createHash('sha256').update(s).digest('hex'),size=s=>Buffer.byteLength(s,'utf8');
  const FILE_BOUND=16*1048576+65536,ENTRY_BOUND=1048576,MAX_RECORDS=16;
  need(typeof file==='string'&&file===resolve(file)&&await fs.realpath(resolve(file,'..'))===resolve(file,'..'),'ANOMALY_JOURNAL_REAL_PARENT_REQUIRED');
  const identityKeys=['cloud_run_id','day_utc','generation','reservation_id','source_head'];
  need(identity&&Object.keys(identity).sort().join('|')===identityKeys.join('|')&&typeof identity.generation==='string'&&identity.generation.length>0&&identity.generation.length<=160&&/^[a-f0-9]{40}$/.test(identity.source_head||'')&&/^[1-9]\d{0,15}$/.test(identity.cloud_run_id||'')&&/^R2RUN:\d+:[a-f0-9]{16}$/.test(identity.reservation_id||'')&&/^\d{4}-\d{2}-\d{2}$/.test(identity.day_utc||''),'ANOMALY_JOURNAL_TRUSTED_IDENTITY_REQUIRED');
  const boundIdentity=JSON.parse(canon(identity)),identity_sha256=hash(canon(boundIdentity));
  const tmp=file+'.next';let state,head=null,broken=false,active_operation=null,recovered_blocked=false;
  const usageCut=u=>u&&['requests','rows_read','rows_written','unknown_ops'].every(k=>Number.isSafeInteger(u[k])&&u[k]>=0)?Object.fromEntries(['requests','rows_read','rows_written','unknown_ops'].map(k=>[k,u[k]])):null;
  const restoredPrepared=e=>{
    need(typeof e.prepared_text==='string'&&size(e.prepared_text)<=131072,'ANOMALY_JOURNAL_PREPARED_BOUND');
    const r=JSON.parse(e.prepared_text),p=prepareAnomalyCommit({logical_event_id:r.logical_event_id,operation_id:r.operation_id,expected_revision:r.expected_revision,card:JSON.parse(r.card_json),descriptor:JSON.parse(r.descriptor_json),plan:JSON.parse(r.plan_json),state:JSON.parse(r.state_json)});
    need(p.text===e.prepared_text&&r.logical_event_id===e.logical_event_id&&r.operation_id===e.operation_id&&r.expected_revision===e.expected_revision&&r.request_sha256===e.request_sha256&&r.first_known_ts===e.first_known_ts,'ANOMALY_JOURNAL_IMMUTABLE_REQUEST_CONFLICT');Object.freeze(p.request);return p;
  };
  const closed=e=>{
    if(e.phase!=='RETURNED'||typeof e.result_text!=='string')return false;
    const r=JSON.parse(e.result_text);let receipt;try{receipt=verifyAnomalyReadback(restoredPrepared(e),JSON.parse(e.readback_reply_text));}catch{return false;}
    return receipt?.status==='COMMITTED_RECEIPT_READBACK'&&r.status==='COMMITTED_RECEIPT_READBACK'&&r.cost_status==='WITHIN_CALLER_PROJECTION'&&r.new_operation_id_allowed!==false&&e.write_reply_text!==null&&e.readback_reply_text!==null&&e.usage_after?.unknown_ops===0;
  };
  const checked=s=>{
    need(s?.schema==='RUNNER_ANOMALY_PENDING_LOCAL_V1'&&canon(s.identity)===canon(boundIdentity)&&s.identity_sha256===identity_sha256&&Number.isSafeInteger(s.sequence)&&s.sequence>=0&&s.sequence<=MAX_RECORDS*6&&Array.isArray(s.records)&&s.records.length<=MAX_RECORDS,'ANOMALY_JOURNAL_IDENTITY_OR_SHAPE_CONFLICT');
    const ids=new Set(),logical=new Set();
    for(const e of s.records){
      need(e&&size(canon(e))<=ENTRY_BOUND&&!ids.has(e.operation_id)&&['PREPARED','WRITE_STARTED','WRITE_REPLY','READBACK_STARTED','READBACK_REPLY','RETURNED'].includes(e.phase),'ANOMALY_JOURNAL_RECORD_CONFLICT');ids.add(e.operation_id);logical.add(e.logical_event_id);restoredPrepared(e);
      for(const [name,max]of [['write_reply_text',65536],['readback_reply_text',196608],['result_text',16384]])need(e[name]===null||typeof e[name]==='string'&&size(e[name])<=max&&canon(JSON.parse(e[name]))===e[name],'ANOMALY_JOURNAL_REPLY_BOUND_OR_ENCODING');
      need((e.phase!=='RETURNED'||e.result_text!==null)&&(!['WRITE_REPLY','READBACK_STARTED','READBACK_REPLY'].includes(e.phase)||e.write_reply_text!==null)&&(e.phase!=='READBACK_REPLY'||e.readback_reply_text!==null),'ANOMALY_JOURNAL_PHASE_REPLY_CONFLICT');
      need(e.usage_before===null||canon(usageCut(e.usage_before))===canon(e.usage_before),'ANOMALY_JOURNAL_USAGE_CONFLICT');
      need(e.usage_after===null||canon(usageCut(e.usage_after))===canon(e.usage_after),'ANOMALY_JOURNAL_USAGE_CONFLICT');
    }
    need(logical.size<=2,'ANOMALY_JOURNAL_LOGICAL_CAPACITY');return s;
  };
  const noPartial=async()=>{try{await fs.lstat(tmp);throw Error('ANOMALY_JOURNAL_UNRESOLVED_NEXT_FILE');}catch(e){if(e.code!=='ENOENT')throw e;}};
  const readExact=async()=>{
    const st=await fs.lstat(file);need(st.isFile()&&!st.isSymbolicLink()&&(st.mode&0o077)===0&&st.size<=FILE_BOUND&&await fs.realpath(file)===file,'ANOMALY_JOURNAL_REAL_FILE_REQUIRED');
    const text=await fs.readFile(file,'utf8');need(size(text)===st.size&&size(text)<=FILE_BOUND,'ANOMALY_JOURNAL_FILE_BOUND');return text;
  };
  const syncParent=async()=>{const h=await fs.open(resolve(file,'..'),'r');try{await h.sync();}finally{await h.close();}};
  const writeHandle=async(name,text)=>{const h=await fs.open(name,'wx',0o600);try{await h.writeFile(text,'utf8');await h.sync();}finally{await h.close();}};
  const encode=s=>{checked(s);const text=canon(s);need(size(text)<=FILE_BOUND,'ANOMALY_JOURNAL_FILE_BOUND');return text;};
  const claimFile=sequence=>file+'.step-'+String(sequence).padStart(3,'0');
  const noNextClaim=async sequence=>{try{await fs.lstat(claimFile(sequence+1));throw Error('ANOMALY_JOURNAL_UNRESOLVED_SEQUENCE_CLAIM');}catch(e){if(e.code!=='ENOENT')throw e;}};
  await noPartial();
  if(initialize_empty){
    await noNextClaim(0);
    need(expected_sha256===null,'ANOMALY_JOURNAL_EXPLICIT_EMPTY_INIT_REQUIRED');
    state={schema:'RUNNER_ANOMALY_PENDING_LOCAL_V1',identity:boundIdentity,identity_sha256,sequence:0,records:[]};
    const text=encode(state);await writeHandle(file,text);await syncParent();need(await readExact()===text,'ANOMALY_JOURNAL_CREATE_READBACK_REQUIRED');head=hash(text);
  }else{
    need(hex(expected_sha256),'ANOMALY_JOURNAL_INDEPENDENT_EXPECTED_SHA_REQUIRED');
    const text=await readExact();need(hash(text)===expected_sha256,'ANOMALY_JOURNAL_EXPECTED_SHA_CONFLICT');state=checked(JSON.parse(text));need(canon(state)===text,'ANOMALY_JOURNAL_CANONICAL_FILE_REQUIRED');head=hash(text);
  }
  await noNextClaim(state.sequence);
  if(state.sequence>0){
    const last=claimFile(state.sequence),st=await fs.lstat(last);need(st.isFile()&&!st.isSymbolicLink()&&(st.mode&0o077)===0&&st.size<=1024,'ANOMALY_JOURNAL_LAST_SEQUENCE_CLAIM_REQUIRED');
    const text=await fs.readFile(last,'utf8'),claim=JSON.parse(text);
    need(text===canon(claim)&&claim.schema==='ANOMALY_PENDING_SEQUENCE_CLAIM_V1'&&claim.sequence===state.sequence&&claim.identity_sha256===identity_sha256&&claim.next_sha256===head&&hex(claim.previous_sha256),'ANOMALY_JOURNAL_SEQUENCE_BINDING_CONFLICT');
  }
  const restored=state.records.filter(e=>!closed(e)).map(e=>JSON.parse(canon(e)));recovered_blocked=restored.length>0;
  const save=async next=>{
    need(!broken,'ANOMALY_JOURNAL_IO_PREVIOUSLY_UNCLOSED');
    try{
      await noPartial();need(hash(await readExact())===head,'ANOMALY_JOURNAL_LOCAL_CAS_CONFLICT');
      const sequence=state.sequence+1,text=encode({...next,sequence});
      // The immutable exclusive claim remains after rename: a delayed writer
      // cannot reuse the same sequence after another writer has removed .next.
      const claim=canon({schema:'ANOMALY_PENDING_SEQUENCE_CLAIM_V1',identity_sha256,sequence,previous_sha256:head,next_sha256:hash(text)});
      await writeHandle(claimFile(sequence),claim);await syncParent();
      need(hash(await readExact())===head,'ANOMALY_JOURNAL_LOCAL_CAS_CONFLICT');
      await writeHandle(tmp,text);await fs.rename(tmp,file);await syncParent();need(await readExact()===text,'ANOMALY_JOURNAL_REPLACE_READBACK_REQUIRED');state=JSON.parse(text);head=hash(text);
    }catch(e){broken=true;throw e;}
  };
  const record=()=>{const e=state.records.find(x=>x.operation_id===active_operation);need(e,'ANOMALY_JOURNAL_ACTIVE_OPERATION_REQUIRED');return e;};
  const update=async delta=>{const e=record();await save({...state,records:state.records.map(x=>x.operation_id===e.operation_id?{...x,...delta}:x)});};
  const restoredResult=e=>{
    const result=e.result_text?JSON.parse(e.result_text):{status:'ANOMALY_PENDING_RESTORED_RECONCILIATION_REQUIRED',new_operation_id_allowed:false,native_admission_qualified:false,event_first_knowledge_qualified:false,native_epoch_qualified:false,SENT_claimed:false};
    if(e.result_had_write_reply&&e.write_reply_text!==null)result.write_reply=JSON.parse(e.write_reply_text);
    if(e.result_had_readback&&e.readback_reply_text!==null)result.retained_readback=JSON.parse(e.readback_reply_text);
    return{...result,operation_id:e.operation_id,request_sha256:e.request_sha256,effective_prepared:restoredPrepared(e),original_origin_revision:e.expected_revision===0?null:e.expected_revision,new_operation_id_allowed:false,restored_pending_phase:e.phase,local_recovery_only:true,known_write_reply:e.write_reply_text===null?null:JSON.parse(e.write_reply_text),known_readback_reply:e.readback_reply_text===null?null:JSON.parse(e.readback_reply_text),retained_prior_usage:e.usage_after??e.usage_before,costs_not_reset:true,automatic_SQL_retry:false};
  };
  const journal={
    identity:Object.freeze(boundIdentity),blocked:()=>broken||recovered_blocked,
    checkpoint:()=>Object.freeze({schema:'RUNNER_ANOMALY_PENDING_CHECKPOINT_V1',identity_sha256,file_sha256:head,sequence:state.sequence,records:state.records.length,private_payload_exported:false,filesystem_durability_qualified:false,native_ACK_qualified:false}),
    restored:()=>restored.map(restoredResult),
    begin:async(prepared,u)=>{
      need(!journal.blocked()&&active_operation===null,'ANOMALY_JOURNAL_RESTORED_OR_ACTIVE_PENDING_BLOCKS_SQL');
      const r=prepared.request;need(!state.records.some(e=>e.operation_id===r.operation_id),'ANOMALY_JOURNAL_OPERATION_ALREADY_RECORDED');
      const e={operation_id:r.operation_id,logical_event_id:r.logical_event_id,expected_revision:r.expected_revision,request_sha256:r.request_sha256,first_known_ts:r.first_known_ts,prepared_text:prepared.text,phase:'PREPARED',write_reply_text:null,readback_reply_text:null,result_text:null,result_had_write_reply:false,result_had_readback:false,usage_before:usageCut(u),usage_after:null};restoredPrepared(e);
      need(state.records.length<MAX_RECORDS&&new Set([...state.records.map(x=>x.logical_event_id),r.logical_event_id]).size<=2,'ANOMALY_JOURNAL_CAPACITY_NO_DROP');await save({...state,records:[...state.records,e]});active_operation=r.operation_id;
    },
    beforeIO:async(payload,u)=>{
      need(!journal.blocked(),'ANOMALY_JOURNAL_RESTORED_PENDING_BLOCKS_SQL');const e=record(),write=payload.sql===COMMIT_SQL;
      need(write?payload.op==='run'&&payload.params?.length===1&&payload.params[0]===e.prepared_text&&e.phase==='PREPARED':payload.sql===READBACK_SQL&&payload.op==='first'&&payload.params?.length===1&&payload.params[0]===e.operation_id&&e.phase==='WRITE_REPLY','ANOMALY_JOURNAL_EXACT_IO_REQUIRED');
      await update({phase:write?'WRITE_STARTED':'READBACK_STARTED',usage_after:usageCut(u)});
    },
    afterIO:async(payload,reply,u)=>{
      const e=record(),write=payload.sql===COMMIT_SQL,text=canon(reply);
      need(e.phase===(write?'WRITE_STARTED':'READBACK_STARTED')&&size(text)<=(write?65536:196608),'ANOMALY_JOURNAL_KNOWN_REPLY_BOUND');
      await update({phase:write?'WRITE_REPLY':'READBACK_REPLY',[write?'write_reply_text':'readback_reply_text']:text,usage_after:usageCut(u)});
    },
    finish:async(result,u)=>{
      const copy={...result};delete copy.effective_prepared;delete copy.write_reply;delete copy.retained_readback;const text=canon(copy);need(size(text)<=16384,'ANOMALY_JOURNAL_RESULT_BOUND');
      await update({phase:'RETURNED',result_text:text,result_had_write_reply:Object.hasOwn(result,'write_reply'),result_had_readback:Object.hasOwn(result,'retained_readback'),usage_after:usageCut(u)});
      if(!closed(record()))recovered_blocked=true;active_operation=null;
    },
  };
  RUNNER_ANOMALY_PENDING_JOURNALS.add(journal);return Object.freeze(journal);
}
